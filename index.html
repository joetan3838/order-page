function buildMinimalistOrderText() { const items = Object.values(cart); if (items.length === 0) return ''; const grouped = []; const brandMap = new Map(); items.forEach(item => { if (!brandMap.has(item.brand)) { const block = { brand: item.brand, lines: [] }; brandMap.set(item.brand, block); grouped.push(block); } brandMap.get(item.brand).lines.push(`${item.name} x ${item.qty}`); }); const blocksText = grouped .map(group => `${group.brand}\n\n${group.lines.join('\n')}`) .join('\n\n'); const totalTypes = items.length; const totalQty = items.reduce((sum, item) => sum + item.qty, 0); return `${blocksText}\n\nTotal: ${totalTypes} types · ${totalQty} pcs`; }
